% =========================================================
% SHARED BOX USER-EDITABLE KEYS
% =========================================================
% x           = left position in inches
% y           = top position in inches
% width       = box width in inches
% height      = box height in inches
% hpad        = left/right padding inside box
% vpad        = top/bottom padding inside box
% angle       = rotation in degrees
% bordercolor = border color
% rule        = border thickness; set rule=0pt for no border
% filename    = file to place in the box
\pgfkeys{
/contentbox/.is family,
/contentbox,
x/.initial=0,
y/.initial=0,
width/.initial=1,
height/.initial=1,
hpad/.initial=\DefaultHPad,
vpad/.initial=\DefaultVPad,
angle/.initial=0,
bordercolor/.initial=\DefaultBorderColor,
rule/.initial=\DefaultBoxRule,
filename/.initial=
}

% =========================================================
% INTERNAL TEXT CONTENT BOX FUNCTION
% =========================================================
% Use this for generated .tex files from Markdown.
%
% filename should point to a .tex file in pages/generated/
\NewDocumentCommand{\ContentBox}{m}{%
  \pgfkeys{/contentbox/.cd,#1}%
  \begin{textblock*}{\pgfkeysvalueof{/contentbox/width}in}
    (\pgfkeysvalueof{/contentbox/x}in,\pgfkeysvalueof{/contentbox/y}in)
    \rotatebox[origin=c]{\pgfkeysvalueof{/contentbox/angle}}{%
      \begin{tcolorbox}[
        width=\pgfkeysvalueof{/contentbox/width}in,
        height=\pgfkeysvalueof{/contentbox/height}in,
        colback=white,
        colframe=\pgfkeysvalueof{/contentbox/bordercolor},
        boxrule=\pgfkeysvalueof{/contentbox/rule},
        arc=\DefaultArc,
        left=\pgfkeysvalueof{/contentbox/hpad},
        right=\pgfkeysvalueof{/contentbox/hpad},
        top=\pgfkeysvalueof{/contentbox/vpad},
        bottom=\pgfkeysvalueof{/contentbox/vpad},
        valign=top,
        halign=flush left,
        sharp corners,
        enhanced
      ]
      \BodyFont
      \input{\GeneratedDir\pgfkeysvalueof{/contentbox/filename}}
      \end{tcolorbox}
    }%
  \end{textblock*}
}

% =========================================================
% INTERNAL IMAGE BOX FUNCTION
% =========================================================
% Use this for PNG/JPG/PDF images stored in pages/content/
%
% filename should point to an image file in pages/content/
%
% The image scales to fit inside the box while preserving aspect ratio.
% For a tall narrow diagram, make the box narrow and tall.
\NewDocumentCommand{\ImageBox}{m}{%
  \pgfkeys{/contentbox/.cd,#1}%
  \begin{textblock*}{\pgfkeysvalueof{/contentbox/width}in}
    (\pgfkeysvalueof{/contentbox/x}in,\pgfkeysvalueof{/contentbox/y}in)
    \rotatebox[origin=c]{\pgfkeysvalueof{/contentbox/angle}}{%
      \begin{tcolorbox}[
        width=\pgfkeysvalueof{/contentbox/width}in,
        height=\pgfkeysvalueof{/contentbox/height}in,
        colback=white,
        colframe=\pgfkeysvalueof{/contentbox/bordercolor},
        boxrule=\pgfkeysvalueof{/contentbox/rule},
        arc=\DefaultArc,
        left=\pgfkeysvalueof{/contentbox/hpad},
        right=\pgfkeysvalueof{/contentbox/hpad},
        top=\pgfkeysvalueof{/contentbox/vpad},
        bottom=\pgfkeysvalueof{/contentbox/vpad},
        valign=center,
        halign=center,
        sharp corners,
        enhanced
      ]
      \includegraphics[
        width=\linewidth,
        height=\tcbtextheight,
        keepaspectratio
      ]{\ContentDir\pgfkeysvalueof{/contentbox/filename}}
      \end{tcolorbox}
    }%
  \end{textblock*}
}
