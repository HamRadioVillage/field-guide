% =========================================================
% DIRECTORIES
% =========================================================
% This is where Pandoc-generated .tex fragments live.
% Content boxes will pull files from this folder.
\newcommand{\GeneratedDir}{pages/generated/}

% =========================================================
% DEFAULT BOX STYLE
% =========================================================
% These are fallback settings used by content boxes.
%
% Default border thickness
% Set rule=0pt in a box definition for NO border.
\newcommand{\DefaultBoxRule}{0.5pt}

% Corner rounding radius
% Set to 0pt for perfectly square corners.
\newcommand{\DefaultArc}{1.2pt}

% Default border color
\newcommand{\DefaultBorderColor}{black}

% Default content padding
\newcommand{\DefaultHPad}{7pt}
\newcommand{\DefaultVPad}{6pt}

% =========================================================
% CONTENT BOX USER-EDITABLE KEYS
% =========================================================
% This is where the available box settings are defined.
%
% x          = left position in inches
% y          = top position in inches
% width      = box width in inches
% height     = box height in inches
% hpad       = left/right padding inside box
% vpad       = top/bottom padding inside box
% angle      = rotation in degrees
% bordercolor= border color
% rule       = border thickness (set to 0pt for no border)
% filename   = generated .tex file to place in the box
\pgfkeys{
/contentbox/.is family,
/contentbox,
x/.initial=0,
y/.initial=0,
width/.initial=1,
height/.initial=1,
hpad/.initial=\DefaultHPad,
vpad/.initial=\DefaultVPad,
angle/.initial=0,
bordercolor/.initial=\DefaultBorderColor,
rule/.initial=\DefaultBoxRule,
filename/.initial=
}

% =========================================================
% INTERNAL CONTENT BOX FUNCTION
% =========================================================
% This is the messy machinery.
% In normal use, you should NOT edit this section unless
% you are changing how the box system fundamentally works.
\NewDocumentCommand{\ContentBox}{m}{%
  \pgfkeys{/contentbox/.cd,#1}%
  \begin{textblock*}{\pgfkeysvalueof{/contentbox/width}in}
    (\pgfkeysvalueof{/contentbox/x}in,\pgfkeysvalueof{/contentbox/y}in)
    \rotatebox[origin=c]{\pgfkeysvalueof{/contentbox/angle}}{%
      \begin{tcolorbox}[
        width=\pgfkeysvalueof{/contentbox/width}in,
        height=\pgfkeysvalueof{/contentbox/height}in,
        colback=white,
        colframe=\pgfkeysvalueof{/contentbox/bordercolor},
        boxrule=\pgfkeysvalueof{/contentbox/rule},
        arc=\DefaultArc,
        left=\pgfkeysvalueof{/contentbox/hpad},
        right=\pgfkeysvalueof{/contentbox/hpad},
        top=\pgfkeysvalueof{/contentbox/vpad},
        bottom=\pgfkeysvalueof{/contentbox/vpad},
        valign=top,
        halign=flush left,
        sharp corners,
        enhanced
      ]
      \BodyFont
      \input{\GeneratedDir\pgfkeysvalueof{/contentbox/filename}}
      \end{tcolorbox}
    }%
  \end{textblock*}
}

\end{document}
