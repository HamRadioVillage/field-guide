% =========================================================
% GENERIC CONTENT BOX
%
% #1 = x position of upper-left corner
% #2 = y position of upper-left corner
% #3 = width
% #4 = height
% #5 = left padding
% #6 = right padding
% #7 = top padding
% #8 = bottom padding
% #9 = markdown filename (relative to \ContentDir)
% #10 = rotation angle (degrees)
%
% Example:
% \ContentBox{0.25}{0.25}{2.5}{3.0}{4pt}{4pt}{4pt}{4pt}{left.md}{90}
% =========================================================
\newcommand{\ContentBox}[10]{%
  \begin{textblock*}{#3in}(#1in,#2in)
    \rotatebox{#10}{%
      \begin{tcolorbox}[
        width=#3in,
        height=#4in,
        colback=white,
        colframe=black,
        boxrule=\DefaultBoxRule,
        arc=\DefaultArc,
        left=#5,
        right=#6,
        top=#7,
        bottom=#8,
        valign=top,
        halign=flush left,
        sharp corners,
        enhanced
      ]
      \markdownInput{\ContentDir#9}
      \end{tcolorbox}
    }
  \end{textblock*}
}
