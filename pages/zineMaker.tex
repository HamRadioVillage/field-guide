% =========================================================
% PLACE ONE TILE
% =========================================================
%
% #1 = slot number
% #2 = x offset from sheet top-left
% #3 = y offset from sheet top-left
%

\newcommand{\PlaceTile}[3]{%
	\edef\ThisPageNum{\GetSlotPage{#1}}%
	\edef\ThisRotation{\GetSlotRotation{#1}}%
	\edef\ThisPageFile{\GetPageFile{\ThisPageNum}}%
	
	% top-left corner of this tile
	\coordinate (TileNW) at ($ (current page.north west) + (#2,-#3) $);
	
	% tile center
	\coordinate (TileCenter) at ($ (TileNW) + (.5\TileWidth,-.5\TileHeight) $);
	
	% placed page
	\node[anchor=center, inner sep=0pt] at (TileCenter) {%
		\includegraphics[
		width=\TileWidth,
		height=\TileHeight,
		keepaspectratio,
		angle=\ThisRotation
		]{\ThisPageFile}%
	};
	
	% debug border
	\draw[gray]
	(TileNW) rectangle ($ (TileNW) + (\TileWidth,-\TileHeight) $);
	
	% debug label
	\node[anchor=north west, font=\tiny]
	at ($ (TileNW) + (2pt,-2pt) $)
	{Slot #1 / Page \ThisPageNum / Rot \ThisRotation};
}
